//@version=6
indicator("Trend Cloud Signals", shorttitle = "Trend Cloud", overlay = true, max_labels_count = 500, max_boxes_count = 500)

// --- Constants ---
const string GROUP_CALC = "Calculation"
const string GROUP_FILTER = "15m Signal Filters"
const string GROUP_VISUAL = "Visuals"
const color BULL_COLOR = #00b915
const color BEAR_COLOR = #e00000
const color TEXT_COLOR = #ffffff

// --- Inputs ---
trendLengthInput = input.int(72, "Trend length", minval = 2, group = GROUP_CALC, tooltip = "Length of the trend cloud center. Higher values filter more noise but produce fewer reversals.")
atrLengthInput = input.int(14, "ATR length", minval = 1, group = GROUP_CALC, tooltip = "ATR length used to size the cloud and trend boundaries.")
trendFactorInput = input.float(2.8, "Trend factor", minval = 0.1, step = 0.1, group = GROUP_CALC, tooltip = "Distance from the cloud center required to recognize a trend reversal. Lower values create more signals.")
cloudWidthInput = input.float(1.25, "Cloud width", minval = 0.1, step = 0.05, group = GROUP_CALC, tooltip = "Width of the displayed cloud in ATR multiples.")

useTrendFilterInput = input.bool(true, "Use EMA direction filter", group = GROUP_FILTER, tooltip = "Only allow BUY signals above a rising EMA and SELL signals below a falling EMA.")
filterEmaLengthInput = input.int(150, "Filter EMA length", minval = 2, group = GROUP_FILTER, tooltip = "EMA used to define the broader directional bias.")
slopeLookbackInput = input.int(5, "EMA slope lookback", minval = 1, group = GROUP_FILTER, tooltip = "The filter EMA must rise or fall over this many bars before a trend signal is accepted.")

useMomentumFilterInput = input.bool(true, "Use RSI filter", group = GROUP_FILTER, tooltip = "Adds RSI momentum confirmation to the signal score.")
rsiLengthInput = input.int(14, "RSI length", minval = 2, group = GROUP_FILTER, tooltip = "RSI calculation length.")
rsiBullLevelInput = input.float(55.0, "RSI BUY threshold", minval = 50.0, maxval = 100.0, step = 0.5, group = GROUP_FILTER, tooltip = "RSI must be at or above this level for bullish confirmation.")
rsiBearLevelInput = input.float(45.0, "RSI SELL threshold", minval = 0.0, maxval = 50.0, step = 0.5, group = GROUP_FILTER, tooltip = "RSI must be at or below this level for bearish confirmation.")

useStrengthFilterInput = input.bool(true, "Use ADX strength filter", group = GROUP_FILTER, tooltip = "Adds ADX directional-strength confirmation to the signal score.")
dmiLengthInput = input.int(14, "DMI length", minval = 1, group = GROUP_FILTER, tooltip = "Directional movement length used by ADX.")
adxSmoothingInput = input.int(14, "ADX smoothing", minval = 1, group = GROUP_FILTER, tooltip = "ADX smoothing length.")
minimumAdxInput = input.float(23.0, "Minimum ADX", minval = 1.0, step = 0.5, group = GROUP_FILTER, tooltip = "ADX must reach this level for strength confirmation.")

useVolatilityFilterInput = input.bool(true, "Use volatility filter", group = GROUP_FILTER, tooltip = "Adds ATR relative-to-price confirmation to the signal score.")
minimumAtrPercentInput = input.float(0.06, "Minimum ATR %", minval = 0.0, step = 0.01, group = GROUP_FILTER, tooltip = "ATR as a percentage of price must reach this level for volatility confirmation.")

useConfirmationScoreInput = input.bool(true, "Use confirmation score", group = GROUP_FILTER, tooltip = "When enabled, RSI, ADX, and volatility act as flexible confirmations instead of requiring every filter to pass.")
minimumConfirmationScoreInput = input.int(2, "Minimum confirmation score", minval = 0, maxval = 3, group = GROUP_FILTER, tooltip = "Required confirmations from RSI, ADX, and volatility. Two is a balanced default; three is more selective; one produces more signals.")
cooldownBarsInput = input.int(10, "Signal cooldown (bars)", minval = 0, group = GROUP_FILTER, tooltip = "Minimum number of candles between accepted signals. Lower values increase signal frequency.")
confirmOnCloseInput = input.bool(true, "Confirm on bar close", group = GROUP_FILTER, tooltip = "Prevents live-bar signals from appearing and disappearing before the candle closes.")

showSignalsInput = input.bool(true, "Show BUY/SELL labels", group = GROUP_VISUAL, tooltip = "Display labels for filtered trend reversals.")
showTakeProfitInput = input.bool(true, "Show TP labels", group = GROUP_VISUAL, tooltip = "Display a TP marker when an opposite filtered trend begins.")
showTrendBackgroundInput = input.bool(true, "Highlight trend changes", group = GROUP_VISUAL, tooltip = "Highlight bars where a filtered trend change is accepted.")
cloudTransparencyInput = input.int(82, "Cloud transparency", minval = 0, maxval = 100, group = GROUP_VISUAL, tooltip = "Transparency of the colored trend cloud.")
showRiskBoxesInput = input.bool(true, "Show SL/TP boxes", group = GROUP_VISUAL, tooltip = "Displays an ATR-based stop-loss box and risk/reward take-profit box for every accepted signal.")
stopLossAtrInput = input.float(1.5, "Stop-loss ATR multiplier", minval = 0.1, step = 0.1, group = GROUP_VISUAL, tooltip = "Distance from the signal close to the stop-loss level, measured in ATR multiples.")
takeProfitRiskRewardInput = input.float(2.0, "Take-profit risk/reward", minval = 0.25, step = 0.25, group = GROUP_VISUAL, tooltip = "Take-profit distance as a multiple of the stop-loss risk. Two means a 1:2 risk/reward target.")
riskBoxBarsInput = input.int(12, "SL/TP box width", minval = 1, maxval = 100, group = GROUP_VISUAL, tooltip = "Number of bars for which each SL and TP box extends to the right.")
riskBoxTransparencyInput = input.int(84, "SL/TP box transparency", minval = 0, maxval = 100, group = GROUP_VISUAL, tooltip = "Transparency of the SL and TP box backgrounds.")

// --- Core calculations ---
basis = ta.ema(close, trendLengthInput)
atrValue = ta.atr(atrLengthInput)
upperTrigger = basis + atrValue * trendFactorInput
lowerTrigger = basis - atrValue * trendFactorInput

var int trend = 1
previousUpperTrigger = upperTrigger[1]
previousLowerTrigger = lowerTrigger[1]
trend := na(previousUpperTrigger) or na(previousLowerTrigger) ? trend : close > previousUpperTrigger ? 1 : close < previousLowerTrigger ? -1 : trend

trendChanged = not na(trend[1]) and trend != trend[1]
bullishChange = trendChanged and trend == 1
bearishChange = trendChanged and trend == -1

// --- Signal filters ---
filterEma = ta.ema(close, filterEmaLengthInput)
rsiValue = ta.rsi(close, rsiLengthInput)
[diPlus, diMinus, adxValue] = ta.dmi(dmiLengthInput, adxSmoothingInput)
atrPercent = close != 0.0 ? atrValue / close * 100.0 : 0.0

emaSlopeBullish = not na(filterEma[slopeLookbackInput]) and filterEma > filterEma[slopeLookbackInput]
emaSlopeBearish = not na(filterEma[slopeLookbackInput]) and filterEma < filterEma[slopeLookbackInput]
trendFilterBullish = not useTrendFilterInput or close > filterEma and emaSlopeBullish
trendFilterBearish = not useTrendFilterInput or close < filterEma and emaSlopeBearish

momentumFilterBullish = not useMomentumFilterInput or rsiValue >= rsiBullLevelInput
momentumFilterBearish = not useMomentumFilterInput or rsiValue <= rsiBearLevelInput
strengthFilter = not useStrengthFilterInput or adxValue >= minimumAdxInput
volatilityFilter = not useVolatilityFilterInput or atrPercent >= minimumAtrPercentInput

// RSI, ADX, and volatility are scored independently. The EMA direction
// filter remains a directional safety filter when it is enabled.
activeConfirmationCount = (useMomentumFilterInput ? 1 : 0) + (useStrengthFilterInput ? 1 : 0) + (useVolatilityFilterInput ? 1 : 0)
requiredConfirmationScore = math.min(minimumConfirmationScoreInput, activeConfirmationCount)
bullishConfirmationScore = (useMomentumFilterInput and rsiValue >= rsiBullLevelInput ? 1 : 0) + (useStrengthFilterInput and adxValue >= minimumAdxInput ? 1 : 0) + (useVolatilityFilterInput and atrPercent >= minimumAtrPercentInput ? 1 : 0)
bearishConfirmationScore = (useMomentumFilterInput and rsiValue <= rsiBearLevelInput ? 1 : 0) + (useStrengthFilterInput and adxValue >= minimumAdxInput ? 1 : 0) + (useVolatilityFilterInput and atrPercent >= minimumAtrPercentInput ? 1 : 0)

confirmationFilterBullish = not useConfirmationScoreInput ? momentumFilterBullish and strengthFilter and volatilityFilter : activeConfirmationCount == 0 or bullishConfirmationScore >= requiredConfirmationScore
confirmationFilterBearish = not useConfirmationScoreInput ? momentumFilterBearish and strengthFilter and volatilityFilter : activeConfirmationCount == 0 or bearishConfirmationScore >= requiredConfirmationScore

barIsConfirmed = not confirmOnCloseInput or barstate.isconfirmed

var int lastSignalBar = na
cooldownPassed = na(lastSignalBar) or bar_index - lastSignalBar >= cooldownBarsInput

filteredBullishChange = bullishChange and barIsConfirmed and cooldownPassed and trendFilterBullish and confirmationFilterBullish
filteredBearishChange = bearishChange and barIsConfirmed and cooldownPassed and trendFilterBearish and confirmationFilterBearish

if filteredBullishChange or filteredBearishChange
    lastSignalBar := bar_index

cloudHalfWidth = atrValue * cloudWidthInput
cloudInnerOffset = cloudHalfWidth * 0.30
cloudOuterOffset = cloudHalfWidth * 1.10
cloudUpper = trend == 1 ? basis - cloudInnerOffset : basis + cloudOuterOffset
cloudLower = trend == 1 ? basis - cloudOuterOffset : basis + cloudInnerOffset
cloudColor = trend == 1 ? BULL_COLOR : BEAR_COLOR

// --- Risk management boxes ---
drawRiskRewardBoxes(const bool isLong) =>
    entryPrice = close
    riskDistance = math.max(atrValue * stopLossAtrInput, syminfo.mintick)
    stopPrice = isLong ? entryPrice - riskDistance : entryPrice + riskDistance
    targetPrice = isLong ? entryPrice + riskDistance * takeProfitRiskRewardInput : entryPrice - riskDistance * takeProfitRiskRewardInput
    rightBar = bar_index + riskBoxBarsInput
    stopTop = math.max(entryPrice, stopPrice)
    stopBottom = math.min(entryPrice, stopPrice)
    targetTop = math.max(entryPrice, targetPrice)
    targetBottom = math.min(entryPrice, targetPrice)
    stopText = "SL\n" + str.tostring(stopPrice, format.mintick)
    targetText = "TP\n" + str.tostring(targetPrice, format.mintick)

    box.new(left = bar_index, top = stopTop, right = rightBar, bottom = stopBottom, xloc = xloc.bar_index, bgcolor = color.new(BEAR_COLOR, riskBoxTransparencyInput), border_color = color.new(BEAR_COLOR, 20), border_style = line.style_dashed, text = stopText, text_size = size.tiny, text_color = TEXT_COLOR, text_halign = text.align_center, text_valign = text.align_center)
    box.new(left = bar_index, top = targetTop, right = rightBar, bottom = targetBottom, xloc = xloc.bar_index, bgcolor = color.new(BULL_COLOR, riskBoxTransparencyInput), border_color = color.new(BULL_COLOR, 20), border_style = line.style_dashed, text = targetText, text_size = size.tiny, text_color = TEXT_COLOR, text_halign = text.align_center, text_valign = text.align_center)

// --- Visual elements ---
plot(basis, "Cloud center", color = color.new(cloudColor, 72), linewidth = 1)
upperCloudPlot = plot(cloudUpper, "Cloud upper edge", color = color.new(cloudColor, 58), linewidth = 1, style = plot.style_linebr)
lowerCloudPlot = plot(cloudLower, "Cloud lower edge", color = color.new(cloudColor, 58), linewidth = 1, style = plot.style_linebr)
fill(upperCloudPlot, lowerCloudPlot, color.new(cloudColor, cloudTransparencyInput), "Trend cloud")

if showSignalsInput and filteredBullishChange
    label.new(bar_index, low, "BUY", xloc = xloc.bar_index, yloc = yloc.belowbar, color = BULL_COLOR, style = label.style_label_up, textcolor = TEXT_COLOR, size = size.small)

if showSignalsInput and filteredBearishChange
    label.new(bar_index, high, "SELL", xloc = xloc.bar_index, yloc = yloc.abovebar, color = BEAR_COLOR, style = label.style_label_down, textcolor = TEXT_COLOR, size = size.small)

if showRiskBoxesInput and filteredBullishChange
    drawRiskRewardBoxes(true)

if showRiskBoxesInput and filteredBearishChange
    drawRiskRewardBoxes(false)

bgcolor(showTrendBackgroundInput and filteredBullishChange ? color.new(BULL_COLOR, 88) : showTrendBackgroundInput and filteredBearishChange ? color.new(BEAR_COLOR, 88) : na, title = "Filtered trend change")

plotshape(showTakeProfitInput and filteredBullishChange, title = "Short take profit", style = shape.labelup, location = location.belowbar, color = color.new(color.gray, 100), text = "TP", textcolor = chart.fg_color, size = size.tiny, offset = -1)
plotshape(showTakeProfitInput and filteredBearishChange, title = "Long take profit", style = shape.labeldown, location = location.abovebar, color = color.new(color.gray, 100), text = "TP", textcolor = chart.fg_color, size = size.tiny, offset = -1)

// --- Alerts ---
alertcondition(filteredBullishChange, "BUY", "Trend Cloud Signals: BUY on {{ticker}} {{interval}}")
alertcondition(filteredBearishChange, "SELL", "Trend Cloud Signals: SELL on {{ticker}} {{interval}}")
